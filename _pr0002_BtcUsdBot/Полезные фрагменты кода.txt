По аналогии с WebSocket  существует протокол EventSource - тут запросы допускаются только в одну сторону - получать потоковую информацию от сервера

// нужно запустить команду: npm install eventsource
// и для использования EventSource вводим след код:
var EventSource = require('eventsource')

// затем код:

let eventSource = new EventSource("/events/subscribe");

eventSource.onmessage = function(event) {
  console.log("Новое сообщение", event.data);
  // этот код выведет в консоль 3 сообщения, из потока данных выше
};

// или eventSource.addEventListener('message', ...)
